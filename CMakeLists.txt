cmake_minimum_required(VERSION 3.22)
project(opendlss_nr_linux LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(UNIX AND NOT APPLE AND CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-stdlib=libc++>)
    add_link_options(-stdlib=libc++)
endif()
file(GLOB NR_HOST_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")
add_executable(dlss5vk ${NR_HOST_SOURCES} tools/volk/volk.c)
target_include_directories(dlss5vk PRIVATE tools/Vulkan-Headers/include tools/volk src)
target_compile_definitions(dlss5vk PRIVATE VK_NO_PROTOTYPES VK_ENABLE_BETA_EXTENSIONS)
target_link_libraries(dlss5vk PRIVATE ${CMAKE_DL_LIBS} pthread)
set_target_properties(dlss5vk PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/build" BUILD_RPATH "$ORIGIN/lib")
add_subdirectory(demo)
add_executable(nr-doctor client/doctor.cpp)
target_include_directories(nr-doctor PRIVATE tools/Vulkan-Headers/include)
target_link_libraries(nr-doctor PRIVATE vulkan)
set_target_properties(nr-doctor PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/build" BUILD_RPATH "$ORIGIN/lib")
